;========================================================
; TdL4 - Cronómetro SS:CC con TMR0
; PIC16F887 - FOSC = 4 MHz (cristal XT)
;========================================================

        LIST    P=16F887
        #include <P16F887.INC>
        ERRORLEVEL -302         ; silenciar avisos de banco

        __CONFIG _CONFIG1, _FOSC_XT & _WDTE_OFF & _PWRTE_ON & _MCLRE_ON & _CP_OFF & _CPD_OFF & _BOREN_OFF & _IESO_OFF & _FCMEN_OFF & _LVP_OFF
        __CONFIG _CONFIG2, _WRT_OFF & _BOR40V


;#define DEBUG
DEBUG_SEG   EQU     .9
DEBUG_CENT  EQU     .95

;--------------------------------------------------------
; CONSTANTES
;--------------------------------------------------------
PRECARGA    EQU     .178        ; precarga de TMR0
TICKS_CENT  EQU     .4          ; interrupciones por centésima
DEB_N       EQU     .8          ; muestras de antirrebote (~20 ms)
LARGA_L     EQU     0x90        ; 400 = 0x0190 (parte baja)
LARGA_H     EQU     0x01        ; 400 = 0x0190 (parte alta)
MASC_SEL    EQU     b'11100001' ; PORTB: bits que NO son selectores (RB1..RB4 = 0)

;--------------------------------------------------------
; VARIABLES (banco 0)
;--------------------------------------------------------
        CBLOCK  0x20
        TICK                    ; interrupciones dentro de la centésima
        DIGITO                  ; dígito que se muestra (0..3)
        CENTESIMAS              ; 0..99
        SEGUNDOS                ; 0..59
        ESTADO                  ; 0=detenido, 1=contando, 2=pausado
        BTN_ESTADO              ; 0=liberado estable, 1=presionado estable
        DEB                     ; contador de antirrebote
        DUR_L                   ; duración de la pulsación (16 bits)
        DUR_H
        LARGA_HECHA             ; 1 si la larga ya se disparó
        EV_CORTO                ; evento pulsación corta (lo pone la ISR)
        EV_LARGO                ; evento pulsación larga (lo pone la ISR)
        D0                      ; buffer de display: decenas de segundos
        D1                      ; unidades de segundos
        D2                      ; decenas de centésimas
        D3                      ; unidades de centésimas (D0..D3 contiguos)
        TMP                     ; auxiliar del multiplexado
        C_COPIA                 ; copias para la conversión (main)
        S_COPIA
        DEC_C
        DEC_S
        ENDC

; Variables de contexto en zona común a todos los bancos (0x70-0x7F)
        CBLOCK  0x70
        W_TEMP
        STATUS_TEMP
        PCLATH_TEMP
        FSR_TEMP
        ENDC

;--------------------------------------------------------
; VECTORES
;--------------------------------------------------------
        ORG     0x0000
        GOTO    INICIO

        ORG     0x0004
        GOTO    ISR

;--------------------------------------------------------
; TABLAS 
;--------------------------------------------------------
; Segmentos, cátodo común. bit0=a ... bit6=g, bit7=dp
TABLA_7SEG:
        ADDWF   PCL,F
        RETLW   b'00111111'     ; 0
        RETLW   b'00000110'     ; 1
        RETLW   b'01011011'     ; 2
        RETLW   b'01001111'     ; 3
        RETLW   b'01100110'     ; 4
        RETLW   b'01101101'     ; 5
        RETLW   b'01111101'     ; 6
        RETLW   b'00000111'     ; 7
        RETLW   b'01111111'     ; 8
        RETLW   b'01101111'     ; 9

; Máscara del selector de cada dígito en PORTB (activo en alto)
TABLA_SEL:
        ADDWF   PCL,F
        RETLW   b'00000010'     ; RB1 (display 1)
        RETLW   b'00000100'     ; RB2 (display 2)
        RETLW   b'00001000'     ; RB3 (display 3)
        RETLW   b'00010000'     ; RB4 (display 4)

;========================================================
; INTERRUPCIÓN (TMR0, cada ~2,5 ms)
;========================================================
ISR:
        ; Guardar contexto
        MOVWF   W_TEMP
        SWAPF   STATUS,W
        MOVWF   STATUS_TEMP
        MOVF    PCLATH,W
        MOVWF   PCLATH_TEMP
        MOVF    FSR,W
        MOVWF   FSR_TEMP
        BCF     STATUS,RP0      ; banco 0
        BCF     STATUS,RP1
        BCF     STATUS,IRP
        CLRF    PCLATH

        BTFSS   INTCON,T0IF
        GOTO    ISR_FIN

        ; Recargar TMR0 y atender la bandera
        MOVLW   PRECARGA
        MOVWF   TMR0
        BCF     INTCON,T0IF

        ; 1) Multiplexado
        CALL    ACTUALIZAR_DISPLAY

        ; 2) Conteo de tiempo (solo ESTADO = 1)
        BTFSS   ESTADO,0
        GOTO    ISR_BOTON

        INCF    TICK,F
        MOVLW   TICKS_CENT
        SUBWF   TICK,W
        BTFSS   STATUS,Z
        GOTO    ISR_BOTON
        CLRF    TICK

        INCF    CENTESIMAS,F
        MOVLW   .100
        SUBWF   CENTESIMAS,W
        BTFSS   STATUS,Z
        GOTO    ISR_BOTON
        CLRF    CENTESIMAS

        INCF    SEGUNDOS,F
        MOVLW   .60
        SUBWF   SEGUNDOS,W
        BTFSC   STATUS,Z
        CLRF    SEGUNDOS        ; 59:99 -> 00:00

        ; 3) Pulsador (RA0 = 0 -> presionado)
ISR_BOTON:
        BTFSC   BTN_ESTADO,0
        GOTO    BOTON_PRESIONADO

        ; --- Estado estable: liberado ---
        BTFSC   PORTA,0
        GOTO    LIBERADO_SUELTO
        INCF    DEB,F           ; RA0=0: posible presión
        MOVLW   DEB_N
        SUBWF   DEB,W
        BTFSS   STATUS,Z
        GOTO    ISR_FIN
        ; Presión validada
        CLRF    DEB
        MOVLW   .1
        MOVWF   BTN_ESTADO
        CLRF    DUR_L
        CLRF    DUR_H
        CLRF    LARGA_HECHA
        GOTO    ISR_FIN

LIBERADO_SUELTO:
        CLRF    DEB             ; rebote o nivel alto estable
        GOTO    ISR_FIN

        ; --- Estado estable: presionado ---
BOTON_PRESIONADO:
        BTFSS   PORTA,0
        GOTO    SIGUE_PRESIONADO

        INCF    DEB,F           ; RA0=1: posible liberación
        MOVLW   DEB_N
        SUBWF   DEB,W
        BTFSS   STATUS,Z
        GOTO    ISR_FIN
        ; Liberación validada
        CLRF    DEB
        CLRF    BTN_ESTADO
        BTFSC   LARGA_HECHA,0   ; si ya hubo larga, no hay corta
        GOTO    ISR_FIN
        MOVLW   .1
        MOVWF   EV_CORTO
        GOTO    ISR_FIN

SIGUE_PRESIONADO:
        CLRF    DEB
        BTFSC   LARGA_HECHA,0   ; la larga actúa una sola vez
        GOTO    ISR_FIN
        INCF    DUR_L,F
        BTFSC   STATUS,Z
        INCF    DUR_H,F
        MOVLW   LARGA_L         ; ¿DUR = 400?
        XORWF   DUR_L,W
        BTFSS   STATUS,Z
        GOTO    ISR_FIN
        MOVLW   LARGA_H
        XORWF   DUR_H,W
        BTFSS   STATUS,Z
        GOTO    ISR_FIN
        MOVLW   .1
        MOVWF   LARGA_HECHA
        MOVWF   EV_LARGO

ISR_FIN:
        MOVF    FSR_TEMP,W
        MOVWF   FSR
        MOVF    PCLATH_TEMP,W
        MOVWF   PCLATH
        SWAPF   STATUS_TEMP,W
        MOVWF   STATUS
        SWAPF   W_TEMP,F
        SWAPF   W_TEMP,W
        RETFIE

;========================================================
; MULTIPLEXADO: apagar anterior, actualizar segmentos,
; habilitar el siguiente. Un dígito por llamada.
; Segmentos en PORTD, selectores en RB1..RB4 (PORTB).
;========================================================
ACTUALIZAR_DISPLAY:
        MOVLW   MASC_SEL        ; 1) apagar los 4 selectores (RB1..RB4 = 0)
        ANDWF   PORTB,F         ;    sin tocar RB0, RB5..RB7

        MOVLW   D0              ; 2) leer el dígito D0+DIGITO
        ADDWF   DIGITO,W
        MOVWF   FSR
        MOVF    INDF,W
        CALL    TABLA_7SEG
        MOVWF   TMP

        MOVF    DIGITO,W        ; punto decimal en el dígito 1
        XORLW   .1
        BTFSC   STATUS,Z
        BSF     TMP,7

        MOVF    TMP,W
        MOVWF   PORTD           ; segmentos nuevos

        MOVF    DIGITO,W        ; 3) habilitar el selector
        CALL    TABLA_SEL
        IORWF   PORTB,F

        INCF    DIGITO,F        ; siguiente dígito (0..3)
        BCF     DIGITO,2
        RETURN

;========================================================
; INICIALIZACIÓN
;========================================================
INICIO:
        BANKSEL ANSEL           ; todo digital (RB1..RB4 son pines analógicos por defecto)
        CLRF    ANSEL
        CLRF    ANSELH

        BANKSEL TRISB
        MOVLW   b'11100001'     ; RB1-RB4 salidas (selectores), resto entradas
        MOVWF   TRISB
        CLRF    TRISD           ; RD0-RD7 salidas (segmentos + dp)
        BSF     TRISA,0         ; RA0 entrada (pulsador)

        ; TMR0: T0CS=0 (FOSC/4), PSA=0 (prescaler en TMR0),
        ; PS=100 (1:32). RBPU=1 (pull-ups de PORTB apagados)
        MOVLW   b'10000100'
        MOVWF   OPTION_REG

        BANKSEL PORTB
        MOVLW   MASC_SEL
        ANDWF   PORTB,F         ; selectores apagados
        CLRF    PORTD           ; segmentos apagados

        CLRF    TICK
        CLRF    DIGITO
        CLRF    CENTESIMAS
        CLRF    SEGUNDOS
        CLRF    ESTADO
        CLRF    BTN_ESTADO
        CLRF    DEB
        CLRF    DUR_L
        CLRF    DUR_H
        CLRF    LARGA_HECHA
        CLRF    EV_CORTO
        CLRF    EV_LARGO
        CLRF    D0
        CLRF    D1
        CLRF    D2
        CLRF    D3

        MOVLW   PRECARGA
        MOVWF   TMR0

        BCF     INTCON,T0IF
        BSF     INTCON,T0IE
        BSF     INTCON,GIE
